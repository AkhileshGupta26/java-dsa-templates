import java.util.*;

class Solution {
    public int totalFruit(int[] nums) {

        int i = 0;
        int j = 0;
        int max = 0;

        HashMap<Integer, Integer> hm = new HashMap<>();

        while (j < nums.length) {

            int a = nums[j];
            hm.put(a, hm.getOrDefault(a, 0) + 1);

            // case 1: less than 2 types
            if (hm.size() < 2) {
                max = Math.max(max, j - i + 1);
                j++;
            }

            // case 2: exactly 2 types
            else if (hm.size() == 2) {
                max = Math.max(max, j - i + 1);
                j++;
            }
      
            // case 3: more than 2 types (INVALID)
            else if (hm.size() > 2) {

                while (hm.size() > 2) {
                    int left = nums[i];
                    hm.put(left, hm.get(left) - 1);
                    
                    //remove from hm if freq ==0
                    if (hm.get(left) == 0) {
                        hm.remove(left);
                    }
                    i++;
                }

                j++; // important
            }
        }

        return max;
    }
}
